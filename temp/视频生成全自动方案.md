### 完整代码示例

```python
# 安装依赖
pip install requests pillow

# Python 脚本
import requests
import base64
import time

# 配置
API_KEY = "你的 API 密钥"
IMAGE_PATH = "cos_photo.jpg"
OUTPUT_PATH = "output_video.mp4"

# 1. 读取图片
with open(IMAGE_PATH, "rb") as f:
    image_data = base64.b64encode(f.read()).decode()

# 2. 调用 API (以 Replicate 为例)
def generate_video(prompt, image_base64):
    url = "https://api.replicate.com/v1/predictions"
    headers = {
        "Authorization": f"Token {API_KEY}",
        "Content-Type": "application/json"
    }
    
    data = {
        "version": "模型版本 ID",
        "input": {
            "prompt": prompt,
            "image": f"data:image/jpeg;base64,{image_base64}",
            "video_length": 5,  # 5 秒
            "fps": 30
        }
    }
    
    # 提交任务
    response = requests.post(url, json=data, headers=headers)
    task = response.json()
    task_id = task["id"]
    
    # 等待完成
    while True:
        time.sleep(5)
        result = requests.get(
            f"{url}/{task_id}",
            headers=headers
        ).json()
        
        if result["status"] == "succeeded":
            return result["output"]
        elif result["status"] == "failed":
            raise Exception("生成失败")

# 3. 生成视频
prompt = """
女仆装少女，温柔微笑，
轻轻整理裙摆，粉色房间，
高质量，动漫风格，5 秒
"""

video_url = generate_video(prompt, image_data)

# 4. 下载视频
video_data = requests.get(video_url).content
with open(OUTPUT_PATH, "wb") as f:
    f.write(video_data)

print(f"视频生成完成：{OUTPUT_PATH}")
```

### 自动化脚本

```python
# batch_generate.py
import os
import subprocess

input_folder = "input_images"
output_folder = "output_videos"

for image in os.listdir(input_folder):
    if image.endswith(('.jpg', '.png')):
        print(f"处理：{image}")
        
        subprocess.run([
            "python", "generate.py",
            "--input", f"{input_folder}/{image}",
            "--output", f"{output_folder}/{image}.mp4",
            "--duration", "5"
        ])
        
        print(f"完成：{image}")

print("批量生成完成！")
```

### 思路

```

4. 脚本自动输入提示词
5. 脚本自动点击下载

使用工具：Selenium / Playwright
```

### 代码示例

```python
# 安装
pip install selenium webdriver-manager

# 自动脚本
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
import time

# 配置
IMAGE_PATH = "cos_photo.jpg"
PROMPT = "女仆装少女，温柔微笑，5 秒"

# 启动浏览器
driver = webdriver.Chrome()
driver.get("https://klingai.com")

# 登录 (需要手动扫码)
input("请扫码登录后按回车继续...")

# 上传图片
upload_btn = driver.find_element(By.CSS_SELECTOR, ".upload-btn")
upload_btn.send_keys(IMAGE_PATH)

# 输入提示词
prompt_input = driver.find_element(By.CSS_SELECTOR, ".prompt-input")
prompt_input.send_keys(PROMPT)

# 点击生成
generate_btn = driver.find_element(By.CSS_SELECTOR, ".generate-btn")
generate_btn.click()

# 等待完成
print("等待生成中...")
time.sleep(300)  # 5 分钟

# 下载
download_btn = driver.find_element(By.CSS_SELECTOR, ".download-btn")
download_btn.click()

print("完成！")
```
